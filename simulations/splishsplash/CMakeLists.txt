cmake_minimum_required(VERSION 3.15)
project(CodexFluidBake LANGUAGES C CXX)
if(NOT SPLISH_SOURCE_DIR)
  message(FATAL_ERROR "Pass -DSPLISH_SOURCE_DIR pointing to the fixed upstream checkout")
endif()
execute_process(COMMAND git -C "${SPLISH_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE SPH_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SPH_REV STREQUAL "dba6b26b9387ab6bdd187d6df0a7c9d490d03500")
  message(FATAL_ERROR "Unexpected SPlisHSPlasH revision")
endif()
set(CMAKE_CXX_STANDARD 11)
set(SPH_LIBS_ONLY ON CACHE BOOL "Core without desktop or Python" FORCE)
set(USE_DOUBLE_PRECISION ON CACHE BOOL "Stable CPU bake" FORCE)
set(USE_AVX OFF CACHE BOOL "Disable explicit AVX solver path" FORCE)
set(USE_THIRD_PARTY_METHODS OFF CACHE BOOL "Selected DFSPH path" FORCE)
set(USE_PYTHON_BINDINGS OFF CACHE BOOL "No Python ABI dependency" FORCE)
set(EIGEN3_INCLUDE_DIR "${SPLISH_SOURCE_DIR}/extern/eigen" CACHE PATH "Bundled Eigen")
list(APPEND CMAKE_MODULE_PATH "${SPLISH_SOURCE_DIR}/CMake")
add_subdirectory("${SPLISH_SOURCE_DIR}" upstream)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(bake_fluid bake_fluid.cpp)
find_package(OpenMP REQUIRED)
target_compile_definitions(bake_fluid PRIVATE USE_DOUBLE USE_CompactNSearch)
target_compile_options(bake_fluid PRIVATE -march=native)
target_include_directories(bake_fluid PRIVATE "${SPLISH_SOURCE_DIR}" "${SPLISH_SOURCE_DIR}/extern/json")
target_link_libraries(bake_fluid PRIVATE SPlisHSPlasH OpenMP::OpenMP_CXX)
add_dependencies(bake_fluid SPlisHSPlasH)
add_executable(sph_kernel_tests "${SPLISH_SOURCE_DIR}/Tests/Kernel/KernelTests.cpp")
target_compile_definitions(sph_kernel_tests PRIVATE USE_DOUBLE USE_CompactNSearch)
target_compile_options(sph_kernel_tests PRIVATE -march=native)
target_include_directories(sph_kernel_tests PRIVATE "${SPLISH_SOURCE_DIR}" "${SPLISH_SOURCE_DIR}/extern/Catch2")
target_link_libraries(sph_kernel_tests PRIVATE SPlisHSPlasH OpenMP::OpenMP_CXX)
add_dependencies(sph_kernel_tests SPlisHSPlasH)
